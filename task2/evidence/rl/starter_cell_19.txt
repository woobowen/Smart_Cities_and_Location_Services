def spacing(F, ideal, nadir):
    """Spacing 指标：归一化后每个解到最近邻的距离的变异系数，越小越均匀"""
    Fn = np.unique((F - ideal) / np.maximum(nadir - ideal, 1e-12), axis=0)
    D = np.abs(Fn[:, None] - Fn[None]).sum(-1)
    np.fill_diagonal(D, np.inf)
    d = D.min(1)
    return d.std() / d.mean()
